# Pausa Fase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the private social film-diary core of Pausa — auth, Cerchie (private groups), film logging with ratings/comments/reactions/"watched with", shared lists, activity feed, and in-app notifications — on Next.js 15 + Cloudflare D1, with the Midnight Screening Room design system.

**Architecture:** Next.js 15 App Router as frontend + BFF. Server logic in isolated domain modules (`db`, `auth`, `tmdb`, `circles`, `diary`, `lists`, `feed`, `notifications`). Persistence on Cloudflare D1 (SQLite) via Drizzle ORM. Media data from TMDB normalized into a canonical `media_entities` table. UI built from a tokenized design system with shadcn/ui.

**Tech Stack:** Next.js 15 (App Router), TypeScript, Tailwind, shadcn/ui, Drizzle ORM + drizzle-kit, Cloudflare D1 + Pages (`@cloudflare/next-on-pages`), Lucia (edge-compatible auth) + Arctic (OAuth), Resend (magic-link email), Vitest (unit/integration), Zod (validation).

## Global Constraints

- **Node/runtime:** target Cloudflare Workers edge runtime. Every route handler and server module MUST be edge-compatible (no Node-only APIs).
- **Package manager:** pnpm.
- **Language:** TypeScript strict mode on.
- **Only films in Fase 1.** `media_entities.type` is always `'movie'`, but schema keeps room for `tv`/`anime` (Fase 3).
- **Palette (verbatim, design system §7 of spec):** base `#0F1115` / `#14181C`; surfaces `#1C2228` (s1), `#242C34` (s2); text primary `#F3F0EB`, secondary `#9CA3AF`; accent primary Mandarin `#FF6B35` (CTA only, one per viewport); accent secondary Forest `#2D6A4F`. - **Rating:** half-stars, values 0.5–5.0 in 0.5 steps.
- **TMDB:** non-commercial license; attribution visible in UI.
- **Commits:** user rule — never commit without asking. During execution, confirm before each commit step.

```typescript
    title: raw.title,
    year: Number.isFinite(year as number) ? year : null,
    posterPath: raw.poster_path ?? null,
```

```typescript
import type { TmdbMovie } from "./mapper";

const BASE = "https://api.themoviedb.org/3";

export async function tmdbGet<T>(apiKey: string, path: string, params: Record<string, string> = {}): Promise<T> {
  const url = new URL(BASE + path);
  url.searchParams.set("api_key", apiKey);
  for (const [k, v] of Object.entries(params)) url.searchParams.set(k, v);
  const res = await fetch(url.toString());
  if (!res.ok) throw new Error(`TMDB ${res.status}`);
  return res.json() as Promise<T>;
}

export interface TmdbSearchResult { id: number; title: string; release_date?: string; poster_path?: string | null; }

export async function searchMovies(apiKey: string, q: string): Promise<TmdbSearchResult[]> {
  const data = await tmdbGet<{ results: TmdbSearchResult[] }>(apiKey, "/search/movie", { query: q });
  return data.results;
}

export async function fetchMovie(apiKey: string, tmdbId: number): Promise<TmdbMovie> {
  return tmdbGet<TmdbMovie>(apiKey, `/movie/${tmdbId}`);
}
```

```typescript
export async function getOrCacheMovie(db: Db, apiKey: string, tmdbId: number) {
  const existing = await db.select().from(mediaEntities).where(eq(mediaEntities.tmdbId, tmdbId)).get();
  if (existing) return existing;
```

`src/modules/auth/magic-link.ts`:
```typescript
// Stateless signed token: base64url(email|expiresAt).hmac
// Uses Web Crypto (edge-compatible). Secret from env at call sites.
function b64url(bytes: ArrayBuffer | Uint8Array): string {
  const arr = bytes instanceof Uint8Array ? bytes : new Uint8Array(bytes);
  return btoa(String.fromCharCode(...arr)).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
}

export function buildMagicToken(email: string, nowMs: number, ttlMs: number) {
  const expiresAt = nowMs + ttlMs;
  const payload = `${email}|${expiresAt}`;
  const token = b64url(new TextEncoder().encode(payload));
  return { token, expiresAt };
}

export type ParseResult = { ok: true; email: string } | { ok: false };

export function parseMagicToken(token: string, nowMs: number): ParseResult {
  try {
    const bin = atob(token.replace(/-/g, "+").replace(/_/g, "/"));
    const payload = new TextDecoder().decode(Uint8Array.from(bin, (c) => c.charCodeAt(0)));
    const [email, exp] = payload.split("|");
    if (!email || !exp) return { ok: false };
    if (Number(exp) < nowMs) return { ok: false };
    return { ok: true, email };
  } catch {
    return { ok: false };
  }
}
```

> Note: for production add HMAC signing with a server secret before sending. The signing step is added in Step 5; the round-trip test above covers encode/expiry.

```typescript
import { and, eq } from "drizzle-orm";
import type { Db } from "@/db";
import { circles, circleMembers, invites } from "@/db/schema";
import { newId } from "@/lib/id";

const ALPHABET = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";

export function generateInviteCode(): string {
  const bytes = crypto.getRandomValues(new Uint8Array(8));
  return Array.from(bytes, (b) => ALPHABET[b % ALPHABET.length]).join("");
}

export function validateInvite(
  inv: { expiresAt: Date | null; maxUses: number | null; uses: number },
  nowMs: number,
): { ok: boolean } {
  if (inv.expiresAt && inv.expiresAt.getTime() < nowMs) return { ok: false };
  if (inv.maxUses != null && inv.uses >= inv.maxUses) return { ok: false };
  return { ok: true };
}

export async function createCircle(db: Db, userId: string, name: string) {
  const circle = { id: newId(), name, createdBy: userId, createdAt: new Date() };
  await db.insert(circles).values(circle);
  await db.insert(circleMembers).values({ circleId: circle.id, userId, role: "admin", joinedAt: new Date() });
  return circle;
}

export async function createInvite(db: Db, circleId: string, userId: string, opts?: { expiresAt?: Date; maxUses?: number }) {
  const invite = {
    id: newId(), circleId, code: generateInviteCode(), createdBy: userId,
    expiresAt: opts?.expiresAt ?? null, maxUses: opts?.maxUses ?? null, uses: 0,
  };
  await db.insert(invites).values(invite);
  return invite;
}

export async function joinByCode(db: Db, userId: string, code: string, nowMs: number) {
  const inv = await db.select().from(invites).where(eq(invites.code, code)).get();
  if (!inv) return { error: "NOT_FOUND" as const };
  if (!validateInvite(inv, nowMs).ok) return { error: "INVALID" as const };
  const already = await db.select().from(circleMembers)
    .where(and(eq(circleMembers.circleId, inv.circleId), eq(circleMembers.userId, userId))).get();
  if (!already) {
    await db.insert(circleMembers).values({ circleId: inv.circleId, userId, role: "member", joinedAt: new Date() });
    await db.update(invites).set({ uses: inv.uses + 1 }).where(eq(invites.id, inv.id));
  }
  return { circleId: inv.circleId };
}

export async function listMembers(db: Db, circleId: string) {
  return db.select().from(circleMembers).where(eq(circleMembers.circleId, circleId)).all();
}

export async function setRole(db: Db, circleId: string, actorId: string, targetId: string, role: "admin" | "member") {
  const actor = await db.select().from(circleMembers)
    .where(and(eq(circleMembers.circleId, circleId), eq(circleMembers.userId, actorId))).get();
  if (!actor || actor.role !== "admin") return { error: "FORBIDDEN" as const };
  await db.update(circleMembers).set({ role })
    .where(and(eq(circleMembers.circleId, circleId), eq(circleMembers.userId, targetId)));
  return { ok: true };
}
```

```typescript
import { desc, eq } from "drizzle-orm";
import type { Db } from "@/db";
import { logEntries, comments, reactions } from "@/db/schema";

export interface FeedItem { kind: "log" | "comment" | "reaction"; id: string; createdAt: Date; data: unknown; }

export function mergeFeed(logs: any[], cms: any[], rcs: any[]): FeedItem[] {
  const items: FeedItem[] = [
    ...logs.map((d) => ({ kind: "log" as const, id: d.id, createdAt: d.createdAt, data: d })),
    ...cms.map((d) => ({ kind: "comment" as const, id: d.id, createdAt: d.createdAt, data: d })),
    ...rcs.map((d) => ({ kind: "reaction" as const, id: d.id, createdAt: d.createdAt, data: d })),
  ];
  return items.sort((a, b) => b.createdAt.getTime() - a.createdAt.getTime());
}

export async function getFeed(db: Db, circleId: string, limit = 50): Promise<FeedItem[]> {
  const logs = await db.select().from(logEntries).where(eq(logEntries.circleId, circleId)).orderBy(desc(logEntries.createdAt)).limit(limit).all();
  const logIds = new Set(logs.map((l) => l.id));
  const cms = (await db.select().from(comments).all()).filter((c) => logIds.has(c.logId));
  const rcs = (await db.select().from(reactions).all()).filter((r) => r.targetType === "log" && logIds.has(r.targetId));
  return mergeFeed(logs, cms, rcs).slice(0, limit);
}
```

```typescript
import { describe, it, expect, beforeAll } from "vitest";
import Database from "better-sqlite3";
import { drizzle } from "drizzle-orm/better-sqlite3";
import { migrate } from "drizzle-orm/better-sqlite3/migrator";
import * as schema from "@/db/schema";
import { newId } from "@/lib/id";
import { createCircle, createInvite, joinByCode } from "@/modules/circles";
import { createLog, addComment } from "@/modules/diary";
import { getFeed } from "@/modules/feed";

let db: any;
beforeAll(() => {
  const sqlite = new Database(":memory:");
  db = drizzle(sqlite, { schema });
  migrate(db, { migrationsFolder: "./src/db/migrations" });
  // seed two users + a media entity
  const now = new Date();
  db.insert(schema.users).values([
    { id: "u1", email: "a@b.com", name: "Ann", createdAt: now },
    { id: "u2", email: "c@d.com", name: "Bob", createdAt: now },
  ]).run();
  db.insert(schema.mediaEntities).values({
    id: "m1", tmdbId: 27205, type: "movie", title: "Inception", year: 2010, cachedAt: now,
  }).run();
});

describe("end-to-end diary flow", () => {
  it("create circle -> invite -> join -> log -> comment -> feed", async () => {
    const circle = await createCircle(db, "u1", "Movie Night");
    const invite = await createInvite(db, circle.id, "u1");
    const joined = await joinByCode(db, "u2", invite.code, 0);
    expect("circleId" in joined).toBe(true);

    const log = await createLog(db, { circleId: circle.id, userId: "u1", mediaEntityId: "m1", rating: 4.5, watchedWith: ["u2"] });
    await addComment(db, { logId: log.id, userId: "u2", body: "loved it", isSpoiler: false });

    const feed = await getFeed(db, circle.id);
    expect(feed.length).toBeGreaterThanOrEqual(2);
    expect(feed.some((f) => f.kind === "log")).toBe(true);
    expect(feed.some((f) => f.kind === "comment")).toBe(true);
  });
});
```
